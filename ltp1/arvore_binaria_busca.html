      Declaração da classe ArvoreBinariaBusca
      
struct No {    int chave;    No* esq;    No* dir;    No(int valor) : chave(valor), esq(nullptr), dir(nullptr) { }}; class ArvoreBinariaBusca {private:    No* raiz;    No* inserirRecursivo(No* no, int valor);    bool buscarRecursivo(No* no, int valor);    void preOrdem(No* no);    void emOrdem(No* no);    void posOrdem(No* no);    void destruir(No* no); public:    ArvoreBinariaBusca() : raiz(nullptr) { }    ~ArvoreBinariaBusca() { destruir(raiz); }    void inserir(int valor);    bool buscar(int valor);    void imprimirPreOrdem();    void imprimirEmOrdem();    void imprimirPosOrdem();};

      
Os métodos recursivos são privados porque recebem um ponteiro para nó, detalhe interno da estrutura. O usuário da classe enxerga apenas os métodos públicos, que sempre começam a recursão a partir de raiz.
